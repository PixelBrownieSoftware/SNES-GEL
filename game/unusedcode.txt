// Box overlap between the player and entity e, using each one's own hitbox.
fn player_touches(u16 e) {
    if (Entities[e].x >= player_x + Entities[0].w) { return 0; }
    if (Entities[e].x + Entities[e].w <= player_x) { return 0; }
    if (Entities[e].y >= player_y + Entities[0].h) { return 0; }
    if (Entities[e].y + Entities[e].h <= player_y) { return 0; }
    return 1;
}
